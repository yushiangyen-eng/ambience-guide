         * 設定分類篩選

        function setCategory(cat) {
            currentCategory = cat;
            document.querySelectorAll('.cat-btn').forEach(btn => {

        /**
         * 執行多重條件篩選並渲染卡片
         */
        function applyFilters() {
            const query = document.getElementById('searchInput').value.trim().toLowerCase();
            const clearBtn = document.getElementById('clearSearchBtn');
            if (query.length > 0) {
                clearBtn.classList.remove('hidden');
            } else {
                clearBtn.classList.add('hidden');
            }

            const filtered = placesData.filter(place => {
                // 1. 分類比對
                const matchCat = currentCategory === 'all' || place.category === currentCategory;
                // 2. 步行時間比對
                const matchWalk = currentWalkMax === 'all' || Number(place.walk_mins) <= Number(currentWalkMax);
                // 3. 關鍵字比對 (同時比對中英文店名、介紹與特色標籤)
                let matchQuery = true;
                if (query) {
                    const searchableText = [
                        place.name_zh,
                        place.name_en,
                        place.tag_zh,
                        place.tag_en,
                        place.desc_zh,
                        place.desc_en
                    ].join(' ').toLowerCase();
                    matchQuery = searchableText.includes(query);
                }

                return matchCat && matchWalk && matchQuery;
            });

            renderCards(filtered);
        }

        /**
         * 完整重新渲染

            updateStaticUIText();
            applyFilters();
        }

        /**
         * ==========================================
         * Google Sheets CSV 動態串接與解析引擎
         * ==========================================
         */
        async function fetchGoogleSheetCSV(url) {
            try {
                const response = await fetch(url);
                if (!response.ok) throw new Error(`HTTP 錯誤狀態碼: ${response.status}`);
                const csvText = await response.text();
                const parsedItems = parseCSVToPlaces(csvText);

                if (parsedItems.length === 0) {
                    throw new Error('CSV 檔案中未解析到有效的店家資料列，請確認欄位標題是否與範本相符。');
                }

                placesData = parsedItems;
                updateDataSourceBadge(true, `已連線 Google 試算表（共 ${parsedItems.length} 筆）`);
                renderAll();
                return { success: true, count: parsedItems.length };
            } catch (error) {
                console.error('載入 Google Sheets CSV 失敗:', error);
                // 若載入失敗，安全降級回內建示範資料，確保房客畫面永不空白
                if (placesData.length === 0) {
                    placesData = [...DEFAULT_PLACES_DATA];
                    renderAll();
                }
                return { success: false, message: error.message };
            }
        }

        /**
         * 輕量級 RFC-4180 相容 CSV 解析器（支援欄位內含逗號、換行與雙引號跳脫）
         */
        function parseCSVToPlaces(csvText) {
            const rows = [];
            let currentRow = [];
            let currentVal = '';
            let insideQuotes = false;

            // 移除可能存在的 UTF-8 BOM
            const cleanText = csvText.replace(/^\uFEFF/, '');

            for (let i = 0; i < cleanText.length; i++) {
                const char = cleanText[i];
                const nextChar = cleanText[i + 1];

                if (char === '"') {
                    if (insideQuotes && nextChar === '"') {
                        currentVal += '"';
                        i++; // 跳過跳脫引號
                    } else {
                        insideQuotes = !insideQuotes;
                    }
                } else if (char === ',' && !insideQuotes) {
                    currentRow.push(currentVal.trim());
                    currentVal = '';
                } else if ((char === '\r' || char === '\n') && !insideQuotes) {
                    if (char === '\r' && nextChar === '\n') i++;
                    currentRow.push(currentVal.trim());
                    if (currentRow.some(cell => cell !== '')) {
                        rows.push(currentRow);
                    }
                    currentRow = [];
                    currentVal = '';
                } else {
                    currentVal += char;
                }
            }
            if (currentVal !== '' || currentRow.length > 0) {
                currentRow.push(currentVal.trim());
                if (currentRow.some(cell => cell !== '')) rows.push(currentRow);
            }

            if (rows.length < 2) return [];

            const headers = rows[0].map(h => h.trim());
            const items = [];

            for (let r = 1; r < rows.length; r++) {
                const rowObj = {};
                headers.forEach((h, idx) => {
                    rowObj[h] = rows[r][idx] || '';
                });

                // 確保至少有中文或英文店名才列入
                if (!rowObj.name_zh && !rowObj.name_en) continue;

                items.push({
                    id: r,
                    category: rowObj.category || 'food',
                    walk_mins: parseInt(rowObj.walk_mins || rowObj.walkMinutes, 10) || 5,
                    name_zh: rowObj.name_zh || rowObj.name_en,
                    name_en: rowObj.name_en || rowObj.name_zh,
                    tag_zh: rowObj.tag_zh || '',
                    tag_en: rowObj.tag_en || rowObj.tag_zh || '',
                    desc_zh: rowObj.desc_zh || rowObj.highlight_zh || '',
                    desc_en: rowObj.desc_en || rowObj.highlight_en || rowObj.desc_zh || '',
                    hours: rowObj.hours || rowObj.hours_zh || '',
                    map_url: rowObj.map_url || rowObj.mapUrl || `https://maps.google.com/?q=${encodeURIComponent(rowObj.name_zh || rowObj.name_en)}`,
                    image_url: rowObj.image_url || ''
                });
            }

            return items;
        }

        /**

        async function loadFromInputUrl() {
            const urlInput = document.getElementById('csvUrlInput').value.trim();
            const statusMsg = document.getElementById('modalStatusMsg');
            
            if (!urlInput) {
                statusMsg.textContent = '請先貼上 Google 試算表發佈的 CSV 網址。';
                statusMsg.className = 'text-xs text-rose-600 block mt-1';
                return;
            }

            statusMsg.textContent = '正在連線讀取試算表資料中...';
            statusMsg.className = 'text-xs text-slate-600 block mt-1';

            const res = await fetchGoogleSheetCSV(urlInput);
            if (res.success) {
                statusMsg.textContent = `✅ 成功載入 ${res.count} 筆店家資料！畫面已即時更新。`;
                statusMsg.className = 'text-xs text-emerald-600 font-medium block mt-1';
                setTimeout(() => closeAdminModal(), 1200);
            } else {
                statusMsg.textContent = `❌ 載入失敗：${res.message}（請確認試算表已「發佈到網路」且格式選為 CSV）`;
                statusMsg.className = 'text-xs text-rose-600 block mt-1';
            }
        }

        /**
         * 下載標準 Google 試算表欄位範本 CSV (含 UTF-8 BOM 確保 Excel 開啟不亂碼)
         */
        function downloadSampleCSV() {
            const headers = [
                'category', 'walk_mins', 'name_zh', 'name_en',
                'tag_zh', 'tag_en', 'desc_zh', 'desc_en',
                'hours', 'map_url', 'image_url'
            ];

            const sampleRows = DEFAULT_PLACES_DATA.map(item => [
                item.category,
                item.walk_mins,
                item.name_zh,
                item.name_en,
                item.tag_zh,
                item.tag_en,
                item.desc_zh,
                item.desc_en,
                item.hours,
                item.map_url,
                item.image_url || ''
            ]);

            const csvContent = '\uFEFF' + [headers, ...sampleRows]
                .map(row => row.map(cell => `"${String(cell).replace(/"/g, '""')}"`).join(','))
                .join('\n');

            const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.download = 'ambience_hotel_guide.csv';
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

        /**
         * XSS 防護跳脫函式
         */
        function escapeHtml(str) {
            if (str === null || str === undefined) return '';
            return String(str)
                .replace(/&/g, '&amp;')
                .replace(/</g, '&lt;')
                .replace(/>/g, '&gt;')
                .replace(/"/g, '&quot;')
                .replace(/'/g, '&#039;');
        }
